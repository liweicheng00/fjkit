{# Feature-local macros for the jobs page. `request` is a parameter, so this file imports without context. #} {% from "ui/button.html" import button %} {% from "ui/data.html" import badge, card, progress %} {% from "ui/feedback.html" import dialog, spinner %} {% from "ui/layout.html" import row %} {% macro job_card(request, job) -%} {% set status %} {% call row(gap=2) %} {# Decorative: the badge beside it already says "Running". #} {% if job.running %}{{ spinner(size="sm", tone="info") }}{% endif %} {{ badge(job.state.value | capitalize, variant=job.state_variant) }} {# Opens the popover and fetches its body. The `id` keeps focus across polls. #} {{ button("Details", variant="ghost", size="xs", icon_name="info", id="job-" ~ job.id ~ "-details", popovertarget="job-detail-" ~ job.id, hx_get=url_for(request, "jobs_detail", job_id=job.id), hx_target="#job-detail-" ~ job.id ~ "-body", hx_swap="innerHTML") }} {% endcall %} {% endset %} {# The card is the poll's target; the hx-* attributes are emitted only while the job runs. #} {% call card(job.label, job.error or (job.processed ~ " of " ~ job.total ~ " steps"), size="sm", actions=status, id="job-" ~ job.id, hx_get=url_for(request, "jobs_status", job_id=job.id) if job.running else none, hx_trigger="load delay:1s" if job.running else none, hx_swap="outerHTML" if job.running else none) %} {{ progress(job.percent) }} {% endcall %} {%- endmacro %} {# The dialog "Details" opens, rendered as a sibling of the card. Title and description are fixed; the body is fetched on open. #} {% macro job_detail_dialog(request, job) -%} {% set footer %} {{ button("Close", variant="outline", size="sm", popovertarget="job-detail-" ~ job.id, popovertargetaction="hide") }} {% endset %} {% call dialog("job-detail-" ~ job.id, title=job.label, description="Background job #" ~ job.id, footer=footer) %} {# A spinner, shown until the body arrives. #} {{ spinner(label="Loading the job detail") }} {% endcall %} {%- endmacro %}
